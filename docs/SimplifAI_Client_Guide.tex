\section*{9. Common tasks (quick)}
\addcontentsline{toc}{section}{Common tasks}

\begin{enumerate}
  \item \textbf{See this month's picture} --- Dashboard $\rightarrow$ set period to Month.
  \item \textbf{Add one bill} --- Transactions $\rightarrow$ Add expense.
  \item \textbf{Load a bank / Excel file} --- Transactions $\rightarrow$ Import from file $\rightarrow$ Confirm.
  \item \textbf{Clear open issues} --- Alerts $\rightarrow$ fix or dismiss each item.
  \item \textbf{Ask a question} --- AI Query $\rightarrow$ type or pick an example.
  \item \textbf{Download for Excel} --- Transactions $\rightarrow$ Export CSV.
\end{enumerate}

\vspace{1.5em}
\noindent\textit{Need a change or found a problem?} Use the feedback (chat) icon on a row or page.
